cmake_minimum_required(VERSION 3.10)
project(airport_motion)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_library(airport_motion SHARED motion.cpp)
target_compile_features(airport_motion PRIVATE cxx_std_17)
target_include_directories(airport_motion PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(airport_motion ${GAZEBO_LIBRARIES})
ament_target_dependencies(airport_motion gazebo_ros rclcpp geometry_msgs nav_msgs)
